<loop>

Iterates over a collection of items for repetitive processing

Control Flow 2.2.0

Overview

The <loop> processor provides iteration capabilities over collections of data. It's essential for processing multiple items, such as lists of products, search results, or any repetitive data structure. The processor automatically handles the iteration and provides access to current item and index variables.

Iterate over XML collections and arrays
Access to current item and index variables
Nested loop support for complex data structures
Conditional processing within loops

Attributes

Attribute Type Required Default Description
item string No item Variable name for current item
index string No index Variable name for current index
maxloops integer No - Maximum number of iterations

Examples

Basic Loop Processing

Simple iteration over a collection of items:

loop-basic-example.xml 
<config xmlns="http://org.webharvest/schema/2.1/core">
  <def var="searchResults">
    <html-to-xml>
      <http url="https://example.com/search?q=products">
      </http>
    </html-to-xml>
  </def>
  
  <!-- Process each search result -->
  <loop item="result" index="i">
    <get var="searchResults"//div[@class='search-result']>
    <def var="resultTitle">
      <xpath expression=".//h3">
        <get var="result"/>
      </xpath>
    </def>
    <def var="resultUrl">
      <xpath expression=".//a/@href">
        <get var="result"/>
      </xpath>
    </def>
    
    <!-- Process each result (index: ${i}) -->
    <template>Processing result ${i}: ${resultTitle} - ${resultUrl}
</template>
  </loop>
</config>

Product Data Processing

Process a list of products with detailed information extraction:

loop-products-example.xml 
<config xmlns="http://org.webharvest/schema/2.1/core">
  <def var="productPage">
    <html-to-xml>
      <http url="https://ecommerce-site.com/products">
      </http>
    </html-to-xml>
  </def>
  
  <!-- Process each product -->
  <loop item="product" index="productIndex">
    <get var="productPage"//div[@class='product-item']>
    
    <!-- Extract product details -->
    <def var="productName">
      <xpath expression=".//h2[@class='product-name']">
        <get var="product"/>
      </xpath>
    </def>
    
    <def var="productPrice">
      <xpath expression=".//span[@class='price']">
        <get var="product"/>
      </xpath>
    </def>
    
    <def var="productImage">
      <xpath expression=".//img/@src">
        <get var="product"/>
      </xpath>
    </def>
    
    <def var="productUrl">
      <xpath expression=".//a[@class='product-link']/@href">
        <get var="product"/>
      </xpath>
    </def>
    
    <!-- Save product data to file -->
    <file path="products/product_${productIndex}.xml" action="write">
      <template><?xml version="1.0" encoding="UTF-8"?>
<product>
  <index>${productIndex}</index>
  <name>${productName}</name>
  <price>${productPrice}</price>
  <image>${productImage}</image>
  <url>${productUrl}</url>
  <scraped_at>${new java.util.Date()}</scraped_at>
</product></template>
    </file>
  </loop>
</config>

Nested Loops

Process hierarchical data with nested loop structures:

loop-nested-example.xml 
<config xmlns="http://org.webharvest/schema/2.1/core">
  <def var="categoryData">
    <html-to-xml>
      <http url="https://store.com/categories">
      </http>
    </html-to-xml>
  </def>
  
  <!-- Process each category -->
  <loop item="category" index="catIndex">
    <get var="categoryData"//div[@class='category']>
    
    <def var="categoryName">
      <xpath expression=".//h2">
        <get var="category"/>
      </xpath>
    </def>
    
    <def var="categoryUrl">
      <xpath expression=".//a/@href">
        <get var="category"/>
      </xpath>
    </def>
    
    <!-- Process products in this category -->
    <loop item="product" index="prodIndex">
      <get var="category"//div[@class='product']>
      
      <def var="productName">
        <xpath expression=".//h3">
          <get var="product"/>
        </xpath>
      </def>
      
      <def var="productPrice">
        <xpath expression=".//span[@class='price']">
          <get var="product"/>
        </xpath>
      </def>
      
      <!-- Output: Category ${catIndex}, Product ${prodIndex} -->
      <template>Category: ${categoryName} | Product: ${productName} | Price: ${productPrice}
</template>
    </loop>
  </loop>
</config>

Advanced Loop Usage

Complex loop processing with conditions and limits:

loop-advanced-example.xml 
<config xmlns="http://org.webharvest/schema/2.1/core">
  <def var="apiData">
    <http url="https://api.example.com/data">
      <http-header name="Accept">application/json</http-header>
    </http>
  </def>
  
  <def var="jsonXml">
    <json-to-xml>
      <get var="apiData"/>
    </json-to-xml>
  </def>
  
  <!-- Process with maximum limit and conditions -->
  <loop item="item" index="i" maxloops="100">
    <get var="jsonXml"//item>
    
    <def var="itemId">
      <xpath expression="id">
        <get var="item"/>
      </xpath>
    </def>
    
    <def var="itemValue">
      <xpath expression="value">
        <get var="item"/>
      </xpath>
    </def>
    
    <def var="itemStatus">
      <xpath expression="status">
        <get var="item"/>
      </xpath>
    </def>
    
    <!-- Only process active items -->
    <if condition="${itemStatus} == 'active'">
      <def var="processedValue">
        <script>
          var value = this.getContext().getVariable("itemValue");
          var index = this.getContext().getVariable("i");
          return "Item " + index + ": " + value.toUpperCase();
        </script>
      </def>
      
      <!-- Log processing -->
      <template>Processing item ${i}: ${itemId} - ${processedValue}
</template>
      
      <!-- Save to database or file -->
      <file path="processed/active_item_${i}.txt" action="write">
        <template>ID: ${itemId}
Value: ${processedValue}
Processed at: ${new java.util.Date()}</template>
      </file>
    </if>
  </loop>
</config>
