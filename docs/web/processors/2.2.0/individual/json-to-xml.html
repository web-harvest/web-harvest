<json-to-xml>

Converts JSON data to XML format for further processing

Data Processing 2.2.0

Overview

The <json-to-xml> processor converts JSON data into XML format, making it compatible with XML-based processors like XPath and XQuery. This is essential when working with REST APIs that return JSON data but you need to process it using XML tools.

Converts JSON objects, arrays, and primitives to XML
Preserves JSON structure and data types
Handles nested objects and arrays
Compatible with all XML processors

Attributes

Attribute Type Required Default Description
id string No - Unique identifier for the processor

Examples

Basic JSON to XML Conversion

Simple conversion of JSON data to XML format:

json-to-xml-basic-example.xml 
<config xmlns="http://org.webharvest/schema/2.1/core">
  <def var="jsonData">
    <http url="https://api.example.com/user/123">
      <http-header name="Accept">application/json</http-header>
    </http>
  </def>
  
  <def var="xmlData">
    <json-to-xml>
      <get var="jsonData"/>
    </json-to-xml>
  </def>
  
  <!-- Now you can use XPath on the XML data -->
  <def var="userName">
    <xpath expression="//name">
      <get var="xmlData"/>
    </xpath>
  </def>
</config>

API Data Processing

Processing REST API responses with JSON-to-XML conversion:

json-to-xml-api-example.xml 
<config xmlns="http://org.webharvest/schema/2.1/core">
  <def var="apiResponse">
    <http url="https://api.github.com/repos/microsoft/vscode">
      <http-header name="Accept">application/vnd.github.v3+json</http-header>
      <http-header name="User-Agent">WebHarvest/2.2.0</http-header>
    </http>
  </def>
  
  <def var="repoXml">
    <json-to-xml>
      <get var="apiResponse"/>
    </json-to-xml>
  </def>
  
  <!-- Extract repository information -->
  <def var="repoName">
    <xpath expression="//name">
      <get var="repoXml"/>
    </xpath>
  </def>
  
  <def var="repoStars">
    <xpath expression="//stargazers_count">
      <get var="repoXml"/>
    </xpath>
  </def>
  
  <def var="repoDescription">
    <xpath expression="//description">
      <get var="repoXml"/>
    </xpath>
  </def>
</config>

JSON Arrays

Processing JSON arrays and converting them to XML:

json-to-xml-arrays-example.xml 
<config xmlns="http://org.webharvest/schema/2.1/core">
  <def var="productsJson">
    <http url="https://api.example.com/products">
      <http-header name="Accept">application/json</http-header>
    </http>
  </def>
  
  <def var="productsXml">
    <json-to-xml>
      <get var="productsJson"/>
    </json-to-xml>
  </def>
  
  <!-- Process each product in the array -->
  <loop item="product" index="i">
    <get var="productsXml"//item>
    
    <def var="productName">
      <xpath expression="name">
        <get var="product"/>
      </xpath>
    </def>
    
    <def var="productPrice">
      <xpath expression="price">
        <get var="product"/>
      </xpath>
    </def>
    
    <def var="productCategory">
      <xpath expression="category">
        <get var="product"/>
      </xpath>
    </def>
    
    <!-- Save each product -->
    <file path="products/product_${i}.xml" action="write">
      <template><?xml version="1.0" encoding="UTF-8"?>
<product>
  <name>${productName}</name>
  <price>${productPrice}</price>
  <category>${productCategory}</category>
</product></template>
    </file>
  </loop>
</config>
