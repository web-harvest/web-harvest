<get var>

Variable Processor 2.2.0

The <get var> processor retrieves the value of a previously defined variable for use in other processors or as output.

Key Features

Variable Retrieval

Retrieve values from previously defined variables

Data Flow

Pass data between processors in complex workflows

Error Handling

Graceful handling of undefined variables

Nested Processing

Use variable values as input for other processors

Attributes

Attribute Type Required Description
var String Yes Name of the variable to retrieve
id String No Unique identifier for the processor

Subprocessors

None

Get var processor does not have subprocessors. It simply retrieves the value of the specified variable.

Examples

Basic Variable Retrieval

basic-retrieval.xml 
<def var="productName">Gaming Laptop</def>
<def var="productPrice">1299.99</def>

<file path="product.txt" action="write">
    <template>
        Product: <get var="productName"/>
        Price: $<get var="productPrice"/>
    </template>
</file>

Data Processing with Variables

data-processing.xml 
<def var="rawHtml">
    <http url="https://example.com/products">
        <http-header name="User-Agent">Web-Harvest Bot</http-header>
    </http>
</def>

<def var="xmlData">
    <html-to-xml>
        <get var="rawHtml"/>
    </html-to-xml>
</def>

<def var="productTitles">
    <xpath expression="//h3[@class='product-title']">
        <get var="xmlData"/>
    </xpath>
</def>

<file path="titles.txt" action="write">
    <get var="productTitles"/>
</file>

Nested Processing Chain

nested-processing.xml 
<def var="searchResults">
    <json-to-xml>
        <http url="https://api.example.com/search?q=laptop">
            <http-header name="Accept">application/json</http-header>
        </http>
    </json-to-xml>
</def>

<def var="firstProduct">
    <xpath expression="//item[1]">
        <get var="searchResults"/>
    </xpath>
</def>

<def var="productDetails">
    <http url="https://api.example.com/products/{id}">
        <http-param name="id">
            <xpath expression="id">
                <get var="firstProduct"/>
            </xpath>
        </http-param>
    </http>
</def>

Complete Data Workflow

complete-workflow.xml 
<def var="baseUrl">https://api.example.com</def>
<def var="apiKey">your-api-key</def>

<def var="categories">
    <json-to-xml>
        <http url="${baseUrl}/categories">
            <http-header name="Authorization">Bearer ${apiKey}</http-header>
        </http>
    </json-to-xml>
</def>

<def var="electronicsId">
    <xpath expression="//category[name='Electronics']/id">
        <get var="categories"/>
    </xpath>
</def>

<def var="products">
    <json-to-xml>
        <http url="${baseUrl}/products?category=${electronicsId}">
            <http-header name="Authorization">Bearer ${apiKey}</http-header>
        </http>
    </json-to-xml>
</def>

<def var="productCount">
    <xpath expression="count(//product)">
        <get var="products"/>
    </xpath>
</def>

<file path="report.xml" action="write">
    <template>
        <![CDATA[
        <report>
            <category>Electronics</category>
            <count>${productCount}</count>
            <products>${products}</products>
        </report>
        ]]>
    </template>
</file>
